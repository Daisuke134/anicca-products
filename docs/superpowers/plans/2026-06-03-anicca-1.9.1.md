```bash
#!/usr/bin/env bash
# Usage: score.sh <json_file> <id1> [id2..]
# Output: JSON array [{id, ai_likelihood, axes}]
#
# 10 axes (Wikipedia "Signs of AI writing"):
#   hedging, repetition, neutrality, formality, lexicalDiversity,
#   templatePhrasing, passiveVoice, uniformParagraphLen,
#   lowSentenceVariety, punctuationOveruse
# Each axis 0-10、 sum = ai_likelihood 0-100

set -euo pipefail
JSON_FILE="${1:?json_file required}"
shift
IDS=("$@")
[[ ${#IDS[@]} -gt 0 ]] || { echo "[]"; exit 0; }

# Use claude haiku for scoring (cheap, 30 tok per quote)
prompt() {
  local id="$1"
  local text
  text=$(jq -r --arg id "$id" '.[$id] // ""' "$JSON_FILE")
  [[ -n "$text" ]] || return
  cat <<EOF
Quote (id=$id): "$text"

10 axes (each 0-10、 higher = more AI-like、 stricter for Japanese):
hedging, repetition, neutrality, formality, lexicalDiversity, templatePhrasing,
passiveVoice, uniformParagraphLen, lowSentenceVariety, punctuationOveruse

Output ONLY JSON object: {"id":"$id","axes":{"hedging":N,...,"punctuationOveruse":N},"ai_likelihood":SUM}
EOF
}

RESULTS=()
for id in "${IDS[@]}"; do
  P=$(prompt "$id") || continue
  RESP=$(echo "$P" | anthropic --model claude-haiku-4-5 --max-tokens 200 2>/dev/null | jq -c '.' 2>/dev/null) || RESP='{"id":"'$id'","axes":{},"ai_likelihood":50}'
  RESULTS+=("$RESP")
done

printf '[%s]\n' "$(IFS=,; echo "${RESULTS[*]}")"
```

```swift
final class PushTokenService {
    private let session: URLSessionProtocol
    init(session: URLSessionProtocol = URLSession.shared) { self.session = session }

    func register(deviceToken: Data) async {
        let token = deviceToken.map { String(format: "%02x", $0) }.joined()
        var request = URLRequest(url: URL(string: "\(API_BASE)/push-tokens")!)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.httpBody = try? JSONSerialization.data(withJSONObject: [
            "deviceToken": token,
            "appVersion": Bundle.main.object(forInfoDictionaryKey: "CFBundleShortVersionString") as? String ?? "",
            "platform": "ios"
        ])

        let delays: [UInt64] = [0, 1_000_000_000, 2_000_000_000]  // 0s, 1s, 2s exponential
        for (attempt, delay) in delays.enumerated() {
            if delay > 0 { try? await Task.sleep(nanoseconds: delay) }
            do {
                let (_, response) = try await session.data(for: request)
                if let http = response as? HTTPURLResponse, (200..<300).contains(http.statusCode) {
                    UserDefaults.standard.set(false, forKey: "pushTokenRegistrationPending")
                    os_log("PushTokenService: registered (attempt %d)", attempt + 1)
                    return
                }
            } catch {
                os_log(.error, "PushTokenService: attempt %d failed: %@", attempt + 1, "\(error)")
            }
        }
        UserDefaults.standard.set(true, forKey: "pushTokenRegistrationPending")
        os_log(.error, "PushTokenService: all 3 attempts failed, flag set")
    }
}
```

import express from 'express';
import { z } from 'zod';
import { rateLimit } from '../../middleware/rateLimit.js';

const schema = z.object({
  email: z.string().regex(/^[^@\s]+@[^@\s]+\.[^@\s]+$/),
  locale: z.string().default('en'),
  deviceId: z.string().min(1)
});

export function newsletterRouter({ resend, prisma, slackAlert }) {
  const router = express.Router();
  router.post('/', rateLimit({ key: r => r.body.deviceId, perMinute: 1, perDay: 1 }), async (req, res) => {
    const parsed = schema.safeParse(req.body);
    if (!parsed.success) return res.status(400).json({ error: 'invalid payload' });
    const { email, locale, deviceId } = parsed.data;

    const existing = await prisma.newsletter_subscribers.findUnique({ where: { device_id: deviceId } });

    // Resend contacts.create with 3 retry
    let contactId = null;
    const SEGMENT_ID = process.env.RESEND_NEWSLETTER_SEGMENT_ID;
    for (let attempt = 1; attempt <= 3; attempt++) {
      try {
        const r = await resend.contacts.create({ email, segments: [{ id: SEGMENT_ID }] });
        contactId = r.data?.id;
        break;
      } catch (e) {
        if (attempt === 3) {
          await prisma.failed_resend_calls.create({ data: { call_type: 'contacts.create', payload_json: { email, deviceId }, error: String(e) } });
          if (slackAlert) await slackAlert(`Resend contacts.create failed 3x: ${e}`);
          return res.status(202).json({ queued: true });
        }
        await new Promise(r => setTimeout(r, 1000 * Math.pow(2, attempt - 1)));
      }
    }

    if (existing) {
      await prisma.newsletter_subscribers.update({
        where: { device_id: deviceId },
        data: { email, locale, resend_contact_id: contactId, opted_out_at: null }
      });
    } else {
      await prisma.newsletter_subscribers.create({
        data: { email, locale, device_id: deviceId, resend_contact_id: contactId }
      });
    }
    res.json({ ok: true });
  });
  return router;
}

import parser from 'cron-parser';

export function parseCronSchedule(expr, from = new Date()) {
  return parser.parseExpression(expr, { currentDate: from }).next().toDate();
}

export async function runNewsletterCron({ resend, anthropic, prisma, slackAlert }) {
  // smoke check first (NB9 mitigation)
  try { await resend.broadcasts.list({ limit: 1 }); }
  catch (e) {
    if (slackAlert) await slackAlert(`Resend broadcasts API check failed: ${e}`);
    return;
  }

  const SEGMENT_ID = process.env.RESEND_NEWSLETTER_SEGMENT_ID;
  const contacts = await prisma.newsletter_subscribers.findMany({ where: { opted_out_at: null } });
  if (contacts.length === 0) return;

  const localeGroups = groupBy(contacts, 'locale');
  for (const [locale, group] of Object.entries(localeGroups)) {
    const affirmation = randomAffirmation(locale);
    let greeting;
    try {
      const r = await anthropic.messages.create({
        model: 'claude-haiku-4-5', max_tokens: 60,
        messages: [{ role: 'user', content: greetingPrompt(locale) }]
      });
      greeting = r.content[0].text;
    } catch {
      greeting = fallbackGreeting(locale);
    }

    const html = renderHtml(greeting, affirmation, locale);
    const subject = subjectFor(locale, affirmation);

    let bcId;
    for (let attempt = 1; attempt <= 5; attempt++) {
      try {
        const r = await resend.broadcasts.create({
          segmentId: SEGMENT_ID,
          audienceId: SEGMENT_ID,  // ★ both per SDK SegmentOptions interface
          from: 'anicca@aniccaai.com',

import express from 'express';
import { z } from 'zod';
import { rateLimit } from '../../middleware/rateLimit.js';

const schema = z.object({
  text: z.string().min(5).max(2000),
  locale: z.string().default('en'),
  appUserId: z.string().optional(),
  appVersion: z.string().optional()
});

export function feedbackRouter({ resend, prisma }) {
  const router = express.Router();
  router.post('/',
    rateLimit({ key: r => r.body.appUserId || r.ip, perMinute: 1, perDay: 10 }),
    async (req, res) => {
      const parsed = schema.safeParse(req.body);
      if (!parsed.success) {
        if (parsed.error.issues.some(i => i.code === 'too_big')) return res.status(413).json({ error: 'too long' });
        return res.status(400).json({ error: 'invalid payload' });
      }
      const { text, locale, appUserId, appVersion } = parsed.data;
      await prisma.feedback_log.create({ data: { text, locale, app_user_id: appUserId, app_version: appVersion } });
      try {
        await resend.emails.send({
          from: 'anicca-feedback@aniccaai.com',
          to: 'keiodaisuke@gmail.com',
          subject: `[Anicca Feedback] ${locale} ${new Date().toISOString()}`,
          text
        });
        res.json({ ok: true });
      } catch {
        res.status(202).json({ queued: true });
      }
    }
  );
  return router;
}
